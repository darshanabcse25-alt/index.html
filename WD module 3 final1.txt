<script>

let qtyInput = document.querySelector("input[type='number']");

qtyInput.addEventListener("input", function () {
    if (this.value > 20) this.value = 20;
    if (this.value < 1) this.value = 1;
});


document.getElementById("nextBtn").addEventListener("click", function () {

    let coffee = document.querySelector("input[name='coffee']:checked");
    let addons = document.querySelectorAll("input[type='checkbox']:checked");
    let qty = parseInt(qtyInput.value);

    let coffeeText = coffee ? coffee.parentElement.innerText.trim() : "Not Selected";

    
    let coffeePrice = 0;
    if (coffeeText.includes("Espresso")) coffeePrice = 30;
    else if (coffeeText.includes("Latte")) coffeePrice = 40;
    else if (coffeeText.includes("Cappuccino")) coffeePrice = 50;

        let addonText = "";
    let addonPrice = 0;

    addons.forEach(a => {
        let name = a.nextSibling.textContent.trim();
        addonText += name + ", ";

        if (name.includes("Milk")) addonPrice += 10;
        if (name.includes("Caramel")) addonPrice += 15;
        if (name.includes("Cream")) addonPrice += 20;
    });

    if (addonText === "") addonText = "None";

    let total = (coffeePrice + addonPrice) * qty;

    
    document.querySelector("#summary p:nth-child(2)").innerHTML =
        "<b>Coffee:</b> " + coffeeText;

    document.querySelector("#summary p:nth-child(3)").innerHTML =
        "<b>Add-ons:</b> " + addonText;

    document.querySelector("#summary p:nth-child(4)").innerHTML =
        "<b>Quantity:</b> " + qty;

    
    let totalTag = document.getElementById("totalPrice");
    if (!totalTag) {
        let p = document.createElement("p");
        p.id = "totalPrice";
        document.querySelector("#summary .box").appendChild(p);
        totalTag = p;
    }

    totalTag.innerHTML = "<b>Total Price:</b> Rs " + total;
});

</script>
